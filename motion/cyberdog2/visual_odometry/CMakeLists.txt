cmake_minimum_required(VERSION 3.10)
project(uran_mivins_odometry)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
find_library(MIVINS_LIBRARY vins_lib)
if(NOT MIVINS_LIBRARY)
  message(FATAL_ERROR "The device MiVINS library is required")
endif()
add_executable(visual_odometry visual_odometry.cpp)
target_compile_features(visual_odometry PRIVATE cxx_std_14)
ament_target_dependencies(visual_odometry rclcpp sensor_msgs nav_msgs message_filters)
target_include_directories(visual_odometry PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(visual_odometry Eigen3::Eigen ${OpenCV_LIBS} ${MIVINS_LIBRARY})
